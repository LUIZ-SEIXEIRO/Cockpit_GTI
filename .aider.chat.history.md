```python
# database_mariadb.py
import os
import mysql.connector # Certifique-se de instalar: pip install mysql-connector-python

# Credenciais do banco de dados lidas do .env
DB_USER = os.getenv('DB_USER')
DB_PASSWORD = os.getenv('DB_PASSWORD')
DB_NAME = os.getenv('DB_NAME')
DB_HOST = os.getenv('DB_HOST', 'localhost')  # Padrão: localhost
DB_PORT = int(os.getenv('DB_PORT', 3306))   # Padrão: 3306

def testar_conexao_mariadb():
    """
    Tenta estabelecer uma conexão com o banco de dados MariaDB usando as credenciais do .env.
    Retorna True se a conexão for bem-sucedida, False caso contrário.
    """
    cnx = None  # Inicializa cnx para garantir que esteja definido
    try:
        print(f"Tentando conectar ao MariaDB em {DB_HOST}:{DB_PORT} com usuário {DB_USER} e banco {DB_NAME}...")
        cnx = mysql.connector.connect(
            user=DB_USER,
            password=DB_PASSWORD,
            host=DB_HOST,
            port=DB_PORT,
            database=DB_NAME,
            connection_timeout=5  # Timeout de 5 segundos para a conexão
        )
        if cnx.is_connected():
            print("Conexão com MariaDB estabelecida com sucesso!")
            return True
    except mysql.connector.Error as err:
        print(f"Erro ao conectar ao MariaDB: {err}")
        return False
    finally:
        if cnx and cnx.is_connected():
            cnx.close()
            print("Conexão com MariaDB fechada.")

```
